import { prisma } from "@/lib/prisma";
import { getCurrentUser } from "@/lib/auth";
import SiteLayout from "@/components/SiteLayout";
import AdminSetUserRoleInline from "@/components/AdminSetUserRoleInline";
import UserIdentity from "@/components/UserIdentity";
import { getRoleLabel } from "@/lib/user-display";

const PAGE_SIZE = 30;



export default async function AdminUsersPage({
  searchParams,
}: {
  searchParams: Promise<{
    page?: string;
    keyword?: string;
  }>;
}) {
  const user = await getCurrentUser();
  const params = await searchParams;

  if (!user) {
    return (
      <SiteLayout title="管理员用户管理" active="admin">
        <div style={panelStyle}>请先登录。</div>
      </SiteLayout>
    );
  }

  if (user.role !== "ADMIN") {
    return (
      <SiteLayout title="管理员用户管理" active="profile">
        <div style={panelStyle}>只有管理员可以访问这个页面。</div>
      </SiteLayout>
    );
  }

  const currentPage = Math.max(1, Number(params.page || "1") || 1);
  const keyword = String(params.keyword || "").trim();

  const where = keyword
    ? {
        OR: [
          {
            nickname: {
              contains: keyword,
            },
          },
          {
            account: {
              contains: keyword,
            },
          },
        ],
      }
    : {};

  const totalCount = await prisma.user.count({ where });
  const totalPages = Math.max(1, Math.ceil(totalCount / PAGE_SIZE));

  const users = await prisma.user.findMany({
    where,
    orderBy: {
      createdAt: "desc",
    },
    skip: (currentPage - 1) * PAGE_SIZE,
    take: PAGE_SIZE,
    include: {
      _count: {
        select: {
          works: true,
          comments: true,
          favorites: true,
          histories: true,
        },
      },
    },
  });

  function buildPageHref(page: number) {
    const usp = new URLSearchParams();
    if (keyword) usp.set("keyword", keyword);
    usp.set("page", String(page));
    return `/admin/users?${usp.toString()}`;
  }

  return (
    <SiteLayout title="管理员用户管理" active="profile">
      <div style={panelStyle}>
        <form
          method="get"
          action="/admin/users"
          style={{
            display: "grid",
            gridTemplateColumns: "1fr 120px",
            gap: 12,
            marginBottom: 20,
          }}
        >
          <input
            name="keyword"
            defaultValue={keyword}
            placeholder="按昵称或账号搜索用户"
            style={inputStyle}
          />

          <button
            type="submit"
            style={{
              border: "none",
              borderRadius: 14,
              background: "#111827",
              color: "#fff",
              fontWeight: 800,
              cursor: "pointer",
            }}
          >
            搜索
          </button>
        </form>

        <div
          style={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            marginBottom: 18,
            gap: 12,
            flexWrap: "wrap",
          }}
        >
          <div style={{ color: "#6b7280", fontSize: 14 }}>
            共 {totalCount} 个用户　/　第 {currentPage} 页，共 {totalPages} 页
          </div>
        </div>

        {users.length === 0 ? (
          <div style={{ color: "#6b7280" }}>没有找到用户</div>
        ) : (
          <div
            style={{
              display: "flex",
              flexDirection: "column",
              gap: 14,
            }}
          >
            {users.map((item) => (
              <div
                key={item.id}
                style={{
                  display: "grid",
                  gridTemplateColumns: "90px 1fr auto",
                  gap: 16,
                  alignItems: "center",
                  border: "1px solid #e5e7eb",
                  borderRadius: 18,
                  padding: 16,
                  background: "#fafafa",
                }}
              >
                <div
                  style={{
                    width: 90,
                    height: 90,
                    borderRadius: "50%",
                    overflow: "hidden",
                    background: "#e5e7eb",
                    border: "1px solid #e5e7eb",
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                  }}
                >
                  {item.avatarUrl ? (
                    <img
                      src={item.avatarUrl}
                      alt={item.nickname}
                      loading="lazy"
                      style={{
                        width: "100%",
                        height: "100%",
                        objectFit: "cover",
                        display: "block",
                      }}
                    />
                  ) : (
                    <span style={{ fontSize: 12, color: "#6b7280" }}>无头像</span>
                  )}
                </div>

                <div>
                  <div style={{ marginBottom: 8 }}>
  <UserIdentity nickname={item.nickname} role={item.role} xp={item.xp} />
</div>

                  <div style={metaStyle}>ID：{item.id}</div>
                  <div style={metaStyle}>账号：{item.account}</div>
                  <div style={metaStyle}>
                    当前身份：{getRoleLabel(item.role)}
                  </div>
                  <div style={metaStyle}>
                    注册时间：{new Date(item.createdAt).toLocaleString()}
                  </div>
                  <div style={metaStyle}>
                    作品：{item._count.works}　评论：{item._count.comments}　收藏：{item._count.favorites}　历史：{item._count.histories}
                  </div>
                </div>

                <AdminSetUserRoleInline
                  account={item.account}
                  currentRole={item.role}
                />
              </div>
            ))}
          </div>
        )}

        {totalPages > 1 && (
          <div
            style={{
              display: "flex",
              gap: 10,
              justifyContent: "center",
              alignItems: "center",
              flexWrap: "wrap",
              marginTop: 24,
            }}
          >
            <a
              href={buildPageHref(Math.max(1, currentPage - 1))}
              style={{
                ...pageBtnStyle,
                pointerEvents: currentPage <= 1 ? "none" : "auto",
                opacity: currentPage <= 1 ? 0.5 : 1,
              }}
            >
              上一页
            </a>

            {Array.from({ length: totalPages }, (_, i) => i + 1)
              .filter((p) => {
                if (totalPages <= 9) return true;
                if (p === 1 || p === totalPages) return true;
                return Math.abs(p - currentPage) <= 2;
              })
              .map((p, idx, arr) => {
                const prev = arr[idx - 1];
                return (
                  <span key={p} style={{ display: "flex", alignItems: "center", gap: 10 }}>
                    {prev && p - prev > 1 && <span style={{ color: "#9ca3af" }}>...</span>}
                    <a
                      href={buildPageHref(p)}
                      style={{
                        ...pageBtnStyle,
                        background: p === currentPage ? "#d4af37" : "#111827",
                      }}
                    >
                      {p}
                    </a>
                  </span>
                );
              })}

            <a
              href={buildPageHref(Math.min(totalPages, currentPage + 1))}
              style={{
                ...pageBtnStyle,
                pointerEvents: currentPage >= totalPages ? "none" : "auto",
                opacity: currentPage >= totalPages ? 0.5 : 1,
              }}
            >
              下一页
            </a>
          </div>
        )}
      </div>
    </SiteLayout>
  );
}

const panelStyle: React.CSSProperties = {
  background: "#fff",
  border: "1px solid #e5e7eb",
  borderRadius: 22,
  padding: 24,
};

const inputStyle: React.CSSProperties = {
  width: "100%",
  padding: 14,
  borderRadius: 14,
  border: "1px solid #d1d5db",
  background: "#fff",
  color: "#111827",
  fontSize: 14,
};

const metaStyle: React.CSSProperties = {
  fontSize: 13,
  color: "#6b7280",
  marginBottom: 6,
};

const pageBtnStyle: React.CSSProperties = {
  padding: "10px 14px",
  borderRadius: 12,
  background: "#111827",
  color: "#fff",
  textDecoration: "none",
  display: "inline-block",
  fontWeight: 700,
  minWidth: 42,
  textAlign: "center",
};
